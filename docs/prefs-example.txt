# Gateway Prefs
#
# Copy this file into the System Preferences folder and name it
# "Gateway Prefs" (no extension). Gateway reads it once at launch.
#
# Lines are "key = value" or "key: value". '#' and ';' start a comment.
# Keys are case-insensitive; surrounding whitespace is trimmed.
# CR, LF and CRLF line endings all work, so it does not matter whether you
# type this on the Mac itself or copy it over from a modern machine.

# ------------------------------------------------------------------ Window
# 1: Gateway opens its log window at launch.
# 0: Gateway starts with no window and just runs.
#
# The File menu is there either way, with Show/Hide Window and Quit, so the
# window can always be brought back and Gateway can always be quit. Hiding the
# window records the choice here, so the next launch starts the same way.
#
# Gateway stays in the Application menu regardless. The Process Manager offers
# exactly one flag for hiding an application (modeOnlyBackground), and it
# removes it from the Application menu and from any dock together -- there is
# no way to ask for one without the other. If a dock such as A-Dock should not
# list Gateway, that is a setting in the dock, not here.
#
# To start Gateway with the Mac, put an alias to it in the Startup Items
# folder inside the System Folder.
show_window    = 1

# ---------------------------------------------------------------- Module 1
# The forward proxy. Point Classilla at this address and port, and turn on
# network.http.proxy.use-http-proxy-for-https so https:// URLs come here too.
http_port      = 8765

# ---------------------------------------------------------------- Module 2
# Ports Outlook Express 5 connects to. SSL stays OFF in Outlook Express for
# all of them; Gateway is the one that speaks TLS.
#
# Configure the account as a plain IMAP or POP account, not as the "Hotmail"
# account type -- that one speaks Microsoft's old HTTPMail protocol, which is
# long dead and is not what Gateway serves.
imap_port      = 1993
pop_port       = 1995
smtp_port      = 1587

# The password you type into Outlook Express. It never leaves this Mac:
# Gateway checks it locally and then authenticates upstream with XOAUTH2.
# Leaving it unset refuses every login.
local_password = change-me

# Upstream mail servers.
#
# IMAP on 993 is TLS from the first byte. SMTP on 587 starts in the clear and
# upgrades with STARTTLS, which Gateway now supports; set smtp_starttls = 0
# and the port to 465 if your provider offers implicit TLS instead.
#
# Note the SMTP host: personal Outlook.com / Hotmail accounts use
# smtp-mail.outlook.com, not smtp.office365.com.
# The hostnames come from the provider setting below, so these are only
# needed to override it.
imap_upstream_port = 993
pop_upstream_port  = 995
smtp_upstream_port = 587
smtp_starttls      = 1

# ---------------------------------------------------------------- Provider
# outlook (the default) or gmail. This one line supplies the OAuth endpoint,
# the scope and the three mail hostnames; anything set explicitly below still
# wins over it.
provider            = outlook
